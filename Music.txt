local isPaused = false
local songButtons = {}
local statusResetThread = nil

-- ============================================
-- HELPER: Update Status dengan reset otomatis
-- ============================================
local function setStatus(text, resetAfter)
    StatusLbl.Text = text
    if statusResetThread then
        task.cancel(statusResetThread)
        statusResetThread = nil
    end
    if resetAfter then
        statusResetThread = task.delay(resetAfter, function()
            if isPlaying and not isPaused then
                StatusLbl.Text = isLooping and "🔁 Sedang diputar (Loop)" or "▶ Sedang diputar"
            end
            statusResetThread = nil
        end)
    end
end

-- ============================================
-- FUNGSI PLAY
-- ============================================
local function PlaySong(index)
    if not Songs[index] then return end
    currentIndex = index
    local song = Songs[index]

    MusicSound:Stop()
    MusicSound.SoundId = song.Id
    MusicSound.TimePosition = 0
    MusicSound.Volume = VolVal

    SongNameLbl.Text = song.Name
    setStatus("⏳ Loading...")

    local ok = pcall(function() MusicSound:Play() end)
    if not ok then
        setStatus("❌ Gagal play (ID ga valid)")
        return
    end

    task.wait(0.3)
    if MusicSound.IsPlaying then
        setStatus(isLooping and "🔁 Sedang diputar (Loop)" or "▶ Sedang diputar")
        PlayBtn.Text = "⏸"
        isPlaying = true
        isPaused = false
    else
        task.wait(1)
        if MusicSound.IsPlaying then
            setStatus(isLooping and "🔁 Sedang diputar (Loop)" or "▶ Sedang diputar")
            PlayBtn.Text = "⏸"
            isPlaying = true
            isPaused = false
        else
            setStatus("❌ Gagal load lagu")
        end
    end

    for i, btn in ipairs(songButtons) do
        if i == index then
            btn.BackgroundColor3 = Theme.AccentD
        else
            btn.BackgroundColor3 = Theme.Bg
        end
    end
end

local function TogglePlay()
    if currentIndex == 0 then
        PlaySong(1)
        return
    end
    if isPlaying and not isPaused then
        MusicSound:Pause()
        PlayBtn.Text = "▶"
        setStatus("⏸ Di-pause")
        isPaused = true
    else
        MusicSound:Resume()
        PlayBtn.Text = "⏸"
        setStatus(isLooping and "🔁 Sedang diputar (Loop)" or "▶ Sedang diputar")
        isPaused = false
    end
end

local function updateLoopVisual()
    if isLooping then
        LoopBtn.BackgroundColor3 = Theme.Good
        LoopBtn.TextColor3 = Color3.new(1, 1, 1)
        loopStroke.Thickness = 1.5
        loopStroke.Transparency = 0.3
        loopStroke.Color = Theme.Good
    else
        LoopBtn.BackgroundColor3 = Theme.Card
        LoopBtn.TextColor3 = Theme.Sub
        loopStroke.Thickness = 0
        loopStroke.Transparency = 1
    end
end

LoopBtn.MouseButton1Click:Connect(function()
    isLooping = not isLooping
    updateLoopVisual()
    if isLooping then
        setStatus("🔁 Loop: ON (lagu ulang terus)", 2)
    else
        setStatus("➡️ Loop: OFF (auto ganti lagu)", 2)
    end
end)
